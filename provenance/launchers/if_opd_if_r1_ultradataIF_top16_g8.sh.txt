# Historical source reference only; use scripts/train.py to launch.
#!/usr/bin/env bash
set -Eeuo pipefail

BASE=${CLUSTER_WORKSPACE}
EXPERIMENT=${CLUSTER_WORKSPACE}/experiments/opd_if_ultradata_20260922
OPD_SRC=${CLUSTER_WORKSPACE}/experiments/opd_six_priority_20260919_run5_isolated
OPD=$OPD_SRC/OPD
PY=$BASE/runtime/old_home/envs/py312/bin/python
LEGACY_SITE=$BASE/runtime/old_home/Doing-XSearcher/.venv/lib/python3.12/site-packages
PAIR=${PAIR:?PAIR required}
RUN_ROOT=$EXPERIMENT/runs/$PAIR
RUN_TAG=${RUN_TAG:?RUN_TAG required}
TEACHER_MODEL=${TEACHER_MODEL:?TEACHER_MODEL required}
GPUS=0,1
N_RESPONSES=8
VAL_FREQ=10
SAVE_FREQ=10
TOTAL_STEPS=200
RAY_TMP=/tmp/opd_if_${PAIR}_ray

export PYTHONPATH="$OPD_SRC/runtime/vllm011:$EXPERIMENT/code:$OPD/verl:$LEGACY_SITE"
unset PYTORCH_CUDA_ALLOC_CONF PYTORCH_ALLOC_CONF
export CUDA_HOME=$BASE/runtime/old_home/envs/py312
export PATH="$CUDA_HOME/bin:$PATH"
export LD_LIBRARY_PATH="$CUDA_HOME/lib${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}"
export LIBRARY_PATH="$CUDA_HOME/lib:$CUDA_HOME/lib64${LIBRARY_PATH:+:$LIBRARY_PATH}"
export TORCH_CUDA_ARCH_LIST=9.0
export HF_ENDPOINT=https://hf-mirror.com HF_HUB_DISABLE_XET=1
export TOKENIZERS_PARALLELISM=true PYTHONUNBUFFERED=1 HYDRA_FULL_ERROR=1
export RAY_memory_usage_threshold=0.99
export CPATH=$LEGACY_SITE/nvidia/curand/include${CPATH:+:$CPATH}
export TORCH_NCCL_BLOCKING_WAIT=1 NCCL_TIMEOUT=7200 NCCL_DEBUG=WARN
export MALLOC_ARENA_MAX=2 MALLOC_TRIM_THRESHOLD_=131072 MALLOC_MMAP_THRESHOLD_=131072
# Credential/account setup omitted from public source reference.
export WANDB_DIR=$RUN_ROOT WANDB_RESUME=allow
export VERL_FILE_LOGGER_PATH=$RUN_ROOT/metrics.jsonl
export NCCL_NVLS_ENABLE=0 OMP_NUM_THREADS=4 MKL_NUM_THREADS=4
export MPLBACKEND=Agg MPLCONFIGDIR=$RUN_ROOT/cache/matplotlib
export HYDRA_FULL_ERROR=1
unset RAY_ADDRESS

export OUTLINES_CACHE_DIR=$RUN_ROOT/cache/outlines
export CUDA_VISIBLE_DEVICES=$GPUS

export ADV_ESTIMATOR=token_reward_direct
export MAX_PROMPT_LENGTH=4096
export MAX_RESP_LENGTH=7168
export MAX_VAL_RESP_LENGTH=16384
export MAX_MODEL_LEN=32768
export MINI_BATCH_SIZE=64
export TEMPERATURE=1.0
export TEACHER_TEMPERATURE=1.0
export REPETITION_PENALTY=1.0
export LOG_PROB_TOP_K=16
export TOP_K_STRATEGY=only_stu
export REWARD_WEIGHT_MODE=student_p
export MODEL_DTYPE=fp32
export LOSS_AGG_MODE=token-mean
export PARALLEL_SIZE=1
export C_PAR_ENABLE=False
export TRAIN_DATASET=${TRAIN_DATASET:-${CLUSTER_WORKSPACE}/experiments/opd_if_ultradata_20260922/data/multi_if/train_turn1.parquet}
export ACTOR_MODEL_PATH=${CLUSTER_WORKSPACE}/experiments/opd_six_priority_20260919_run5_isolated/models/DeepSeek-R1-Distill-Qwen-1.5B
export REWARD_MODEL_PATH=$TEACHER_MODEL
export CKPT_PATH=$RUN_ROOT/checkpoints/$RUN_TAG
export VAL_PATH=$RUN_ROOT/val_log/$RUN_TAG
PPO_MAX_TOKEN_LEN_PER_GPU=32768
NGPU=$(awk -F, '{print NF}' <<< "$GPUS")
TEST_DATASET='["${CLUSTER_WORKSPACE}/experiments/opd_if_ultradata_20260922/data/multi_if/val_turn1_sub800.parquet"]'

mkdir -p "$RUN_ROOT" "$OUTLINES_CACHE_DIR" "$CKPT_PATH" "$VAL_PATH"
cd "$OPD"

"$PY" "$EXPERIMENT/code/train_entry.py" \
  algorithm.adv_estimator=$ADV_ESTIMATOR \
  algorithm.grpo_outcome_weight=1.0 \
  actor_rollout_ref.nccl_timeout=7200 \
  data.shuffle=False \
  data.train_files="$TRAIN_DATASET" \
  data.val_files="$TEST_DATASET" \
  data.train_batch_size=$MINI_BATCH_SIZE \
  data.max_prompt_length=$MAX_PROMPT_LENGTH \
  data.max_response_length=$MAX_RESP_LENGTH \
  data.filter_overlong_prompts=True \
  data.truncation=error \
  data.return_raw_chat=True \
  +data.apply_chat_template_kwargs.enable_thinking=False \
  actor_rollout_ref.model.path=$ACTOR_MODEL_PATH \
  actor_rollout_ref.model.use_remove_padding=True \
  actor_rollout_ref.model.enable_activation_offload=True \
  actor_rollout_ref.model.enable_gradient_checkpointing=True \
  actor_rollout_ref.actor.optim.lr=1e-6 \
  actor_rollout_ref.actor.ppo_mini_batch_size=$MINI_BATCH_SIZE \
  actor_rollout_ref.actor.use_dynamic_bsz=True \
  actor_rollout_ref.actor.ppo_micro_batch_size_per_gpu=1 \
  actor_rollout_ref.actor.ppo_max_token_len_per_gpu=$PPO_MAX_TOKEN_LEN_PER_GPU \
  actor_rollout_ref.actor.ulysses_sequence_parallel_size=$PARALLEL_SIZE \
  actor_rollout_ref.actor.use_kl_loss=False \
  actor_rollout_ref.actor.loss_agg_mode=$LOSS_AGG_MODE \
  actor_rollout_ref.actor.c_par.enable=$C_PAR_ENABLE \
  actor_rollout_ref.actor.c_par.freq=10 \
  actor_rollout_ref.actor.c_par.symmetric=False \
  actor_rollout_ref.actor.fsdp_config.param_offload=False \
  actor_rollout_ref.actor.fsdp_config.optimizer_offload=False \
  actor_rollout_ref.actor.fsdp_config.forward_prefetch=True \
  actor_rollout_ref.actor.fsdp_config.model_dtype=$MODEL_DTYPE \
  actor_rollout_ref.rollout.max_num_batched_tokens=$PPO_MAX_TOKEN_LEN_PER_GPU \
  actor_rollout_ref.ref.fsdp_config.param_offload=True \
  actor_rollout_ref.ref.fsdp_config.model_dtype=$MODEL_DTYPE \
  actor_rollout_ref.ref.log_prob_use_dynamic_bsz=True \
  actor_rollout_ref.rollout.name=vllm \
  actor_rollout_ref.rollout.temperature=$TEMPERATURE \
  actor_rollout_ref.rollout.log_prob_use_dynamic_bsz=True \
  +actor_rollout_ref.rollout.log_prob_top_k=$LOG_PROB_TOP_K \
  +actor_rollout_ref.rollout.top_k_strategy=$TOP_K_STRATEGY \
  +actor_rollout_ref.rollout.reward_weight_mode=$REWARD_WEIGHT_MODE \
  +actor_rollout_ref.rollout.teacher_temperature=$TEACHER_TEMPERATURE \
  actor_rollout_ref.rollout.tensor_model_parallel_size=$PARALLEL_SIZE \
  actor_rollout_ref.rollout.gpu_memory_utilization=0.5 \
  actor_rollout_ref.rollout.free_cache_engine=True \
  actor_rollout_ref.rollout.max_model_len=$MAX_MODEL_LEN \
  actor_rollout_ref.rollout.n=$N_RESPONSES \
  actor_rollout_ref.rollout.val_kwargs.do_sample=True \
  +actor_rollout_ref.rollout.val_kwargs.max_tokens=$MAX_VAL_RESP_LENGTH \
  actor_rollout_ref.rollout.val_kwargs.n=8 \
  actor_rollout_ref.rollout.val_kwargs.temperature=0.6 \
  actor_rollout_ref.rollout.val_kwargs.top_p=0.95 \
  actor_rollout_ref.rollout.repetition_penalty=$REPETITION_PENALTY \
  actor_rollout_ref.rollout.calculate_log_probs=True \
  actor_rollout_ref.ref.log_prob_micro_batch_size_per_gpu=1 \
  reward_model.enable=True \
  +reward_model.reward_kwargs.enable_format_reward=False \
  reward_model.model.path=$REWARD_MODEL_PATH \
  reward_model.model.input_tokenizer=null \
  reward_model.model.use_remove_padding=True \
  reward_model.model.fsdp_config.param_offload=True \
  +reward_model.model.dtype=$MODEL_DTYPE \
  reward_model.micro_batch_size_per_gpu=24 \
  custom_reward_function.path=${CLUSTER_WORKSPACE}/experiments/opd_if_ultradata_20260922/code/if_reward.py \
  custom_reward_function.name=reward_func \
  trainer.val_before_train=False \
  trainer.log_val_generations=2 \
  trainer.logger='["console","wandb","file"]' \
  trainer.project_name=opd_analysis \
  trainer.experiment_name=$RUN_TAG \
  trainer.validation_data_dir=$VAL_PATH \
  trainer.n_gpus_per_node=$NGPU \
  trainer.nnodes=1 \
  trainer.save_freq=$SAVE_FREQ \
  trainer.test_freq=-1 \
  trainer.total_epochs=10 \
  trainer.total_training_steps=$TOTAL_STEPS \
  trainer.resume_mode=auto \
  trainer.default_local_dir=$CKPT_PATH \
  trainer.is_plot=False \
  trainer.max_actor_ckpt_to_keep=null \
  ray_kwargs.ray_init.num_cpus=24 \
  +ray_kwargs.ray_init.num_gpus=2 \
  +ray_kwargs.ray_init.address=local \
  +ray_kwargs.ray_init._temp_dir=$RAY_TMP \
  +ray_kwargs.ray_init.include_dashboard=False \
  hydra.run.dir=$RUN_ROOT/hydra \
  "$@"
